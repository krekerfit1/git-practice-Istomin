# 1. БАЗОВАЯ НАСТРОЙКА И ИНИЦИАЛИЗАЦИЯ
---
git version
  — Проверка установленной версии Git на компьютере.

git config --global user.name "Имя"
git config --global user.email "email@example.com"
  — Глобальная настройка имени и почты автора для подписи всех коммитов.

git init -b main
  — Инициализация нового локального Git-репозитория с основной веткой main.


# 2. РАБОТА С ИЗМЕНЕНИЯМИ И КОММИТАМИ
---
git status
  — Просмотр текущего состояния репозитория.

git add <имя_файла>
  — Добавление конкретного файла в индекс перед коммитом.

git add .
  — Добавление всех изменённых и новых файлов в индекс.

git commit -m "Сообщение коммита"
  — Фиксация подготовленных изменений в истории репозитория с поясняющим комментарием.


# 3. ПРОСМОТР ИСТОРИИ И АНАЛИЗ ИЗМЕНЕНИЙ
---
git log
  — Подробный просмотр истории коммитов (полный хеш, автор, дата, сообщение). Выход — q.

git log --oneline
  — Компактный вывод истории коммитов.

git show <хеш_коммита>
  — Детальный просмотр конкретного коммита.


# 4. УПРАВЛЕНИЕ ВЕТКАМИ
---
git branch
  — Просмотр списка локальных веток (текущая активная ветка отмечается звёздочкой *).

git branch -a
  — Просмотр всех веток .

git branch <имя_ветки>
  — Создание новой ветки без переключения на неё.

git checkout -b <имя_ветки>
  — Создание новой ветки и немедленное переключение на неё .

git switch -c <имя_ветки>
  — Создание новой ветки и переключение на неё.

git switch <имя_ветки>
  — Переключение на существующую ветку.

git switch -
  —  переключение обратно на предыдущую ветку.

git branch -d <имя_ветки>
  — Безопасное удаление локальной ветки.

git branch -D <имя_ветки>
  — Принудительное удаление локальной ветки.


# 5. РАБОТА С УДАЛЁННЫМ РЕПОЗИТОРИЕМ
-------------------------------------------------------------------
git remote add origin <URL_репозитория>
  — Привязка удалённого репозитория под стандартным псевдонимом origin.

git remote -v
  — Проверка подключенных удалённых репозиториев (адреса для fetch и push).

git push -u origin main
  — Отправка коммитов локальной ветки main на сервер и настройка отслеживания (tracking) между main и origin/main.

git push --set-upstream origin <имя_ветки>
  — Первая отправка новой локальной ветки в удалённый репозиторий с привязкой к удалённой ветке.

git pull
  — Получение изменений из удалённого репозитория и их автоматическое слияние с текущей локальной веткой.

git fetch --prune
  — Обновление информации об удалённых ветках с удалением локальных ссылок на ветки, которые были удалены на сервере.


# 6. ПРИМЕРЫ ИЗ ЛАБОРАТОРНОЙ РАБОТЫ
---
### Инициализация и привязка к GitHub:
git init -b main
dotnet new console 
git remote add origin https://github.com/Username/Lab2-Git_FIO.git
git add .
git commit -m "Initial commit: create console app"
git push -u origin main

### Разработка в отдельной ветке feature и отправка для Pull Request:
git switch -c feature
git add .
git commit -m "new branch"
git push --set-upstream origin feature

### Синхронизация и очистка после слияния Pull Request на GitHub:
git switch main
git pull
git branch -D feature
git fetch --prune
